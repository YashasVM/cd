async function authenticated(url, token, options = {}) {

function download(url, token) {
  const parts = [];
  let firstAt = null;
  let received = 0;
  let response;
  let request;
  const done = new Promise((resolve, reject) => {
    request = httpRequest(url, { headers: authorization(token) }, incoming => {
      response = incoming;
      if (incoming.statusCode !== 200) { incoming.resume(); reject(new Error(`download status ${incoming.statusCode}`)); return; }
      incoming.on('data', part => { firstAt ??= performance.now(); parts.push(part); received += part.length; });
      incoming.once('end', () => resolve({ parts, bytes: received, firstAt, completedAt: performance.now() }));
      incoming.once('error', reject);
    });
    request.once('error', reject);
    request.setTimeout(10000, () => request.destroy(new Error('download deadline')));
    request.end();
  });
  done.catch(() => {});
  return { done, abort: () => { response?.destroy(); request.destroy(); } };
}
async function upload(url, token, body) {
  let request;
  const done = new Promise((resolve, reject) => {
    request = httpRequest(url, { method: 'POST', headers: { ...authorization(token), 'Content-Length': body.length, 'Content-Type': 'application/octet-stream' } }, response => {
      response.resume();
      response.once('end', () => resolve({ status: response.statusCode, completedAt: performance.now() }));
      response.once('error', reject);
    });
    request.once('error', reject);
    request.setTimeout(10000, () => request.destroy(new Error('upload deadline')));
  });
  done.catch(() => {});
  for (let at = 0; at < body.length; at += 768 * 1024) {
    if (!request.write(body.subarray(at, Math.min(body.length, at + 768 * 1024)))) await once(request, 'drain');
  }
  request.end();
  return done;
}
async function websocketTrial(base, sender, receiver) {
  const socketURL = role => { const url = new URL(`ws/${role}`, base); url.protocol = 'ws:'; return url; };
  const incoming = new WebSocket(socketURL('download'), { headers: authorization(receiver), perMessageDeflate: false });
  let outgoing;
  try {
    const parts = []; let bytes = 0, firstAt = null;
    const done = new Promise((resolve, reject) => {
      incoming.on('message', (data, binary) => {
        if (!binary && data.toString() === 'end') { resolve({ parts, bytes, firstAt, completedAt: performance.now() }); return; }
        if (!binary) { reject(new Error('unexpected WebSocket control')); return; }
        firstAt ??= performance.now(); parts.push(data); bytes += data.length;
      });
      incoming.once('error', reject);
      incoming.once('close', code => { if (code !== 1000) reject(new Error(`receiver WebSocket closed ${code}`)); });
    });
    done.catch(() => {});
    await once(incoming, 'open');
    outgoing = new WebSocket(socketURL('upload'), { headers: authorization(sender), perMessageDeflate: false });
    await once(outgoing, 'open');
    const started = performance.now();
    for (let at = 0; at < payload.length; at += 768 * 1024) {
      const part = payload.subarray(at, Math.min(payload.length, at + 768 * 1024));
      while (outgoing.bufferedAmount + part.length > 4 * 1048576) await yieldLoop();
      outgoing.send(part, { binary: true });
    }
    outgoing.send('end');
    let timer;
    const result = await Promise.race([done, new Promise((_, reject) => { timer = setTimeout(() => reject(new Error('WebSocket deadline')), 10000); })]).finally(() => clearTimeout(timer));
    const ms = result.completedAt - started;
    assert.equal(result.bytes, payload.length); const receivedSHA256 = sha(Buffer.concat(result.parts)); assert.equal(receivedSHA256, expected);
    return { bytes: result.bytes, receivedSHA256, elapsedMs: ms, firstByteMs: result.firstAt - started, mibPerSecond: mib / (ms / 1000) };
  } finally { incoming.terminate(); outgoing?.terminate(); }
}
